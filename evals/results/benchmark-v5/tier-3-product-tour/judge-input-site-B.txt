(function () {
  var root = document.documentElement;
  var STEPS = ['intro', 'features', 'pricing', 'start'];

  function stepOf(url) {
    try {
      var m = new URL(url, location.href).pathname.match(/\/(features|pricing|start)(\/|\/index\.html|\.html)?$/);
      return m ? STEPS.indexOf(m[1]) : 0;
    } catch (e) { return -1; }
  }
  var here = stepOf(location.href);

  if (/Android/i.test(navigator.userAgent)) root.classList.add('is-android');
  root.classList.add('js');
  // Billing choice from Pricing (monthly | yearly), applied before first paint so nothing jumps.
  try { root.dataset.billing = sessionStorage.getItem('tally-billing') || 'monthly'; } catch (e) { root.dataset.billing = 'monthly'; }

  // Shared with tour.js: resolves when the incoming transition is done (or immediately).
  var resolveReady;
  window.__tourReady = new Promise(function (r) { resolveReady = r; });
  window.__tourArrived = false; // true when we came from another tour page

  function setDir(fromUrl) {
    var from = fromUrl ? stepOf(fromUrl) : -1;
    root.dataset.dir = from > here ? 'back' : 'forward';
    return from;
  }

  // Old page: tag the outgoing snapshot with the same direction.
  window.addEventListener('pageswap', function (e) {
    if (!e.viewTransition || !e.activation || !e.activation.entry) return;
    var to = stepOf(e.activation.entry.url);
    root.dataset.dir = to < here ? 'back' : 'forward';
    if (e.viewTransition.types) e.viewTransition.types.add(root.dataset.dir);
  });

  // New page: decide direction before the first rendered frame.
  window.addEventListener('pagereveal', function (e) {
    var act = window.navigation && navigation.activation;
    var fromUrl = act && act.from ? act.from.url : (document.referrer || '');
    var sameSite = fromUrl && fromUrl.indexOf(location.origin) === 0;
    var from = sameSite ? setDir(fromUrl) : -1;
    window.__tourArrived = from >= 0 && from !== here;
    if (e.viewTransition) {
      if (e.viewTransition.types) e.viewTransition.types.add(root.dataset.dir || 'forward');
      e.viewTransition.finished.finally(function () { resolveReady(true); });
    } else {
      resolveReady(false);
    }
  });

  // Browsers without pagereveal: ready on DOM load, direction from referrer.
  if (!('onpagereveal' in window)) {
    var ref = document.referrer;
    if (ref && ref.indexOf(location.origin) === 0) {
      var f = setDir(ref);
      window.__tourArrived = f >= 0 && f !== here;
    }
    document.addEventListener('DOMContentLoaded', function () { resolveReady(false); });
  }
})();

(function () {
  var reduce = window.matchMedia('(prefers-reduced-motion: reduce)');
  var ready = window.__tourReady || Promise.resolve(false);

  
  ready.then(function () {
    if (!window.__tourArrived) return;          // first visit / shared link: leave focus alone
    var h = document.getElementById('tour-title');
    if (h) h.focus({ preventScroll: true });
  });

  
  document.addEventListener('keydown', function (e) {
    if (e.defaultPrevented || e.altKey || e.ctrlKey || e.metaKey || e.shiftKey) return;
    var t = e.target;
    if (t.closest && t.closest('input, textarea, select, [role="group"], [contenteditable]')) return;
    var link = e.key === 'ArrowRight' ? document.querySelector('a.nav-next:not([aria-hidden])')
             : e.key === 'ArrowLeft' ? document.querySelector('a.nav-back:not([aria-hidden])') : null;
    if (link) { e.preventDefault(); link.click(); }
  });

  
  var players = {};
  function mount(slot) {
    if (!window.lottie) return;
    var holder = slot.querySelector('.anim-player');
    var anim = window.lottie.loadAnimation({
      container: holder, renderer: 'svg', loop: false, autoplay: false,
      path: slot.dataset.src,
      rendererSettings: { preserveAspectRatio: 'xMidYMid meet', progressiveLoad: false }
    });
    players[slot.dataset.name] = anim;
    anim.addEventListener('DOMLoaded', function () {
      var last = anim.totalFrames - 1;
      var mode = slot.dataset.mode; // "toggle" = driven by the billing control
      if (mode === 'toggle') {
        var yearly = slot.dataset.state === 'yearly';
        anim.goToAndStop(yearly ? last : 0, true);
        slot.classList.add('is-live');
        return;
      }
      if (reduce.matches) {
        anim.goToAndStop(last, true);
        slot.classList.add('is-live');
        return;
      }
      ready.then(function () {
        anim.goToAndStop(0, true);
        slot.classList.add('is-live');
        anim.play();
      });
    });
    anim.addEventListener('data_failed', function () { slot.classList.remove('is-live'); });
  }
  function boot() { document.querySelectorAll('.anim[data-src]').forEach(mount); }
  if (window.lottie) boot(); else window.addEventListener('load', boot);

  
  var KEY = 'tally-billing';
  function getBilling() { try { return sessionStorage.getItem(KEY) || 'monthly'; } catch (e) { return 'monthly'; } }
  function setBilling(v) { try { sessionStorage.setItem(KEY, v); } catch (e) {} }

  var group = document.querySelector('.billing');
  if (group) {
    var slot = document.querySelector('.anim[data-name="toggle"]');
    var apply = function (v, animate) {
      group.querySelectorAll('button').forEach(function (b) { b.setAttribute('aria-pressed', String(b.value === v)); });
      document.documentElement.dataset.billing = v;
      document.querySelectorAll('.toggle-labels [data-for]').forEach(function (el) { el.dataset.on = String(el.dataset.for === v); });
      if (slot) slot.dataset.state = v;
      var a = players.toggle;
      if (a && a.isLoaded) {
        var last = a.totalFrames - 1;
        if (!animate || reduce.matches) a.goToAndStop(v === 'yearly' ? last : 0, true);
        else a.playSegments(v === 'yearly' ? [0, last] : [last, 0], true);
      }
      setBilling(v);
    };
    group.addEventListener('click', function (e) {
      var b = e.target.closest('button');
      if (b) apply(b.value, true);
    });
    group.addEventListener('keydown', function (e) {
      if (e.key !== 'ArrowLeft' && e.key !== 'ArrowRight') return;
      e.preventDefault();
      var v = e.key === 'ArrowRight' ? 'yearly' : 'monthly';
      apply(v, true);
      group.querySelector('button[value="' + v + '"]').focus();
    });
    apply(getBilling(), false);
  }

})();
